## Desafio - location /saude/ com URI no proxy_pass e /api2/ sem barra final
   (arquivo editado: nginx.conf - '        # Proxy reverso: tudo em /api/ é encaminhado para o container "backend"' -> '        # Desafio: proxy_pass COM URI (/status) - o trecho que casou com a
        # location (/saude/) é SUBSTITUÍDO por /status no destino.
        location /saude/ {
            proxy_pass http://backend:4000/status;
        }

        # Comparação: proxy_pass SEM URI (sem barra final) - o caminho
        # original é repassado inteiro, com o prefixo /api2/.
        location /api2/ {
            proxy_pass http://backend:4000;
        }

        # Proxy reverso: tudo em /api/ é encaminhado para o container "backend"')
